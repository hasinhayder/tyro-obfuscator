QUICKSTART

        

          ONE-LINE INTEGRATION 

          IN VITE.CONFIG.JS
        

        

          Drop it into your existing Laravel + Vite configuration alongside laravel-vite-plugin:
        


        

          

            vite.config.js
            ESM / CJS
          

          
import { defineConfig } from 'vite';
import laravel from 'laravel-vite-plugin';
import tyroObfuscator from 'tyro-obfuscator';

export default defineConfig({
  plugins: [
    laravel([
      'resources/css/app.css',
      'resources/js/app.js',
    ]),

    // Add Tyro Obfuscator — runs automatically on npm run build
    tyroObfuscator({
      preset: 'balanced', // 'light' | 'balanced' | 'aggressive'
    }),
  ],
});
